HÀM VẼ NHÂN VẬT (Khối vuông đỏ viền vàng tại tọa độ X, Y) function drawCharacter(x, y) { let size = 90; // Kích thước khối vuông // Vẽ khối vuông đỏ ctx.fillStyle = "#ff0000"; ctx.fillRect(x - size / 2, y - size / 2, size, size); // Vẽ viền vàng phát sáng ctx.strokeStyle = "#ffff00"; ctx.lineWidth = 4; ctx.strokeRect(x - size / 2, y - size / 2, size, size); // Vẽ tâm chấm trắng chính giữa ctx.beginPath(); ctx.arc(x, y, 6, 0, 2 * Math.PI, false); ctx.fillStyle = "#ffffff"; ctx.fill(); } // 2. HÀM XỬ LÝ CAMERA & QUÉT TỌA ĐỘ MÀU XANH LÁ function processKhoi3() { if (!webcamElement) { findAndShareCamera(); } if (webcamElement && webcamElement.readyState >= 2) { canvas.width = window.innerWidth; canvas.height = window.innerHeight; // Vẽ khung hình từ Camera lên Canvas (giữ nguyên không lật) ctx.drawImage(webcamElement, 0, 0, canvas.width, canvas.height); // Đọc dữ liệu điểm ảnh (Pixels) const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height); const data = imgData.data; let sumX = 0; let sumY = 0; let count = 0; let step = 6; // Quét cách bước để chạy mượt, không lag // Duyệt từng pixel tìm màu Xanh Lá for (let i = 0; i < data.length; i += 4 * step) { let r = data[i]; let g = data[i + 1]; let b = data[i + 2]; // Điều kiện: Chỉ cần kênh Xanh Lá (G) sáng hơn Xanh Dương (B) và Đỏ (R) if (g > 20 && g > r && g > b) { let pixelIndex = i / 4; let x = pixelIndex % canvas.width; let y = Math.floor(pixelIndex / canvas.width); sumX += x; sumY += y; count++; } } // Nếu tìm thấy mảng màu xanh lá (ít nhất 3 pixels) if (count > 3) { // Tính tọa độ trung tâm X, Y let avgX = Math.round(sumX / count); let avgY = Math.round(sumY / count); // GỌI HÀM VẼ NHÂN VẬT TẠI TỌA ĐỘ BẮT ĐƯỢC drawCharacter(avgX, avgY); // Cập nhật bảng thông số HUD statusText.innerText = "⚡ ĐÃ KHÓA MỤC TIÊU"; statusText.style.color = "#00ffcc"; coordsText.innerText = `Tọa độ (X, Y): X = ${avgX} px | Y = ${avgY} px`; statsText.innerText = `Diện tích: ${count} px`; } else { // Không thấy màu xanh statusText.innerText = "🔍 ĐANG TÌM ĐÈN..."; statusText.style.color = "#ff0055"; coordsText.innerText = `Tọa độ (X, Y): -- , --`; statsText.innerText = `Diện tích: 0 px`; } } // Lặp lại liên tục requestAnimationFrame(processKhoi3); }
