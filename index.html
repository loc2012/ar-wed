// ========================================================================= // KHỐI 3: VẼ NHÂN VẬT & QUÉT TỌA ĐỘ MÀU XANH LÁ (FULL ĐẦY ĐỦ) // ========================================================================= // 1. HÀM CHUYÊN VẼ NHÂN VẬT TẠI TỌA ĐỘ (X, Y) function drawCharacter(x, y) { let size = 90; // Kích thước khối vuông nhân vật // Vẽ Khối vuông đỏ ctx.fillStyle = "#ff0000"; ctx.fillRect(x - size / 2, y - size / 2, size, size); // Vẽ Viền vàng phát sáng ctx.strokeStyle = "#ffff00"; ctx.lineWidth = 4; ctx.strokeRect(x - size / 2, y - size / 2, size, size); // Vẽ Tâm chấm trắng chính giữa ctx.beginPath(); ctx.arc(x, y, 6, 0, 2 * Math.PI, false); ctx.fillStyle = "#ffffff"; ctx.fill(); } // 2. HÀM VÒNG LẬP CHÍNH: QUÉT CAMERA, TÍNH TỌA ĐỘ VÀ VẼ NHÂN VẬT function processKhoi3() { // Kiểm tra & Khởi động Camera nếu chưa bật if (!webcamElement) { findAndShareCamera(); } if (webcamElement && webcamElement.readyState >= 2) { // Khớp kích thước Canvas với màn hình thiết bị canvas.width = window.innerWidth; canvas.height = window.innerHeight; // Render khung hình camera lên Canvas (giữ nguyên không lật) ctx.drawImage(webcamElement, 0, 0, canvas.width, canvas.height); // Trích xuất dữ liệu điểm ảnh (Pixels) const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height); const data = imgData.data; let sumX = 0; let sumY = 0; let count = 0; let step = 6; // Bước nhảy để tối ưu hiệu năng, tránh giật lag // --- THUẬT TOÁN QUÉT MÀU XANH SIÊU NHẠY --- for (let i = 0; i < data.length; i += 4 * step) { let r = data[i]; let g = data[i + 1]; let b = data[i + 2]; // Tính phần trăm kênh màu Xanh Lá (G) let total = r + g + b; if (total === 0) total = 1; // Chống lỗi chia cho 0 let percentG = (g / total) * 100; // ĐIỀU KIỆN: Chỉ cần kênh G > 15, chiếm trên 35% tổng màu và sáng hơn R, B if (g > 15 && percentG > 35 && g > r && g > b) { let pixelIndex = i / 4; let x = pixelIndex % canvas.width; let y = Math.floor(pixelIndex / canvas.width); sumX += x; sumY += y; count++; } } // --- XỬ LÝ KẾT QUẢ QUÉT MÀU --- if (count > 3) { // 1. Tính tọa độ trung tâm X, Y của vùng màu xanh let avgX = Math.round(sumX / count); let avgY = Math.round(sumY / count); // 2. GỌI HÀM VẼ NHÂN VẬT TẠI TỌA ĐỘ ĐÃ XÁC ĐỊNH drawCharacter(avgX, avgY); // 3. Cập nhật Bảng thông số HUD trên màn hình statusText.innerText = "⚡ ĐÃ KHÓA MỤC TIÊU"; statusText.style.color = "#00ffcc"; coordsText.innerText = `Tọa độ (X, Y): X = ${avgX} px | Y = ${avgY} px`; statsText.innerText = `Diện tích: ${count} px`; } else { // Không tìm thấy màu xanh trong khung hình statusText.innerText = "🔍 ĐANG TÌM ĐÈN..."; statusText.style.color = "#ff0055"; coordsText.innerText = `Tọa độ (X, Y): -- , --`; statsText.innerText = `Diện tích: 0 px`; } } // Tạo vòng lặp liên tục redraw theo tần số quét màn hình requestAnimationFrame(processKhoi3); }
